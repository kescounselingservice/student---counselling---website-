const COUNSELLORS = { "VAINAVI MAM": { displayName: "Vainavi Ma'am" }, "KINNARI MAM": { displayName: "Kinnari Ma'am" } }; window.addEventListener('DOMContentLoaded', () => { const savedCounsellor = sessionStorage.getItem("activeCounsellor"); if (savedCounsellor && COUNSELLORS[savedCounsellor]) { showDashboard(COUNSELLORS[savedCounsellor].displayName); } else { window.location.href = "index.html"; } }); function showDashboard(displayName) { document.getElementById("CounsellorTitle").innerText = displayName; renderBookings(displayName); } function getBookings() { const possibleKeys = ["mindspace_bookings"]; for (const key of possibleKeys) { const data = localStorage.getItem(key); if (data) { try { return { storageKey: key, list: JSON.parse(data) }; } catch (e) { console.error("Error parsing bookings data", e); } } } return { storageKey: null, list: [] }; } function renderBookings(displayName) { const container = document.getElementById("bookingsList"); const { storageKey, list } = getBookings(); if (!storageKey || list.length === 0) { container.innerHTML = '

No pending or confirmed sessions found.

'; return; } const relevant = list.filter(b => b.counsellor === "Group Support" || b.counsellor === "Individual Counsellor"); if (relevant.length === 0) { container.innerHTML = '

No counseling sessions found.

'; return; } container.innerHTML = relevant.map((b, index) => { const origIndex = list.findIndex(item => item === b); let statusBadge = ''; let actionBtn = ''; if (b.status === "confirmed") { statusBadge = 'Confirmed'; actionBtn = ` `; } else if (b.status === "rescheduled") { statusBadge = 'Rescheduled'; actionBtn = ` `; } else if (b.status === "declined") { statusBadge = 'Declined'; actionBtn = ` `; } else { // Pending statusBadge = 'Pending Approval'; actionBtn = ` `; } return `

${b.fullName} | Student

${b.date}
${statusBadge}
Service:${b.service}
Date & Time:${b.date} at ${b.time}
Notes:
${b.notes || 'N/A'}
${actionBtn}
`; }).join(''); } function updateBookingStatus(storageKey, index, newStatus, displayName) { try { const raw = localStorage.getItem(storageKey); if (!raw) return; const list = JSON.parse(raw); if (list[index]) { list[index].status = newStatus; list[index].approvedBy = displayName; localStorage.setItem(storageKey, JSON.stringify(list)); renderBookings(displayName); } } catch (e) { console.error(`Failed to update booking to ${newStatus}`, e); } } function approveBooking(storageKey, index, displayName) { updateBookingStatus(storageKey, index, "confirmed", displayName); } function rescheduleBooking(storageKey, index, displayName) { updateBookingStatus(storageKey, index, "rescheduled", displayName); } function declineBooking(storageKey, index, displayName) { updateBookingStatus(storageKey, index, "declined", displayName); } function logout() { sessionStorage.removeItem("activeCounsellor"); sessionStorage.removeItem("activeCounsellor"); localStorage.removeItem("activeCounsellor"); window.location.href = "index.html"; }
